#include "driver/gpio.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#define MOTOR_EN_PIN   GPIO_NUM_18
#define ENCODER_A_PIN  GPIO_NUM_34

static void motor_task(void *arg)
{
    gpio_set_direction(MOTOR_EN_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(ENCODER_A_PIN, GPIO_MODE_INPUT);

    while (1) {
        // PID loop keeps wheel speed locked to setpoint
        int32_t ticks = encoder_read_ticks();
        pid_update(&loop, ticks, DT_MS);
        vTaskDelay(pdMS_TO_TICKS(10));
    }
}

void app_main(void)
{
    xTaskCreate(motor_task, "motor_task", 4096, NULL, 5, NULL);
}
